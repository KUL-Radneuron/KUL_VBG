#   python3 - <<'PY' > fsl-packages-full.txt
#   import json, glob
#   names = {json.load(open(f)).get('name') for f in glob.glob('$FSLDIR/conda-meta/*.json')
#            if 'fslconda' in (json.load(open(f)).get('channel') or '')}
#   print('\n'.join(sorted(n for n in names if n)))
#   PY
